💻 Спецификация API программного модуля

Система предоставляет автономный класс ImageEnhancerAPI для асинхронного запуска задач в Web Worker с поддержкой методов постановки, получения статуса, прерывания и получения результата, а также событий изменения статуса.

1. Описание методов и пример использования

import { ImageEnhancerAPI } from './js/api/imageEnhancerAPI.js';

const api = new ImageEnhancerAPI();

// Подписка на событие изменения статуса задачи
api.onStatusChange((event) => {
    console.log(`Задача ${event.taskId} -> Статус: ${event.status} (${event.progress}%)`);
});

// 1. Метод постановки задачи (принимает File, Blob, ImageData или HTMLImageElement)
const taskId = await api.submitTask(fileInput.files[0]);

// 2. Метод получения статуса задачи
const status = api.getStatus(taskId);
console.log(status.progress); // Прогресс от 0 до 100

// 3. Метод прерывания задачи
// api.abortTask(taskId);

// 4. Метод получения готового изображения
const resultBlob = await api.getResult(taskId, 'blob');
imageElement.src = URL.createObjectURL(resultBlob);
